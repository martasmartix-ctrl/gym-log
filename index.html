import 'package:ultimate_gym_tracker/models/pr_record.dart'; import 'package:ultimate_gym_tracker/models/workout.dart'; import 'package:uuid/uuid.dart'; class StorageService extends ChangeNotifier { late Box _exercisesBox; late Box _workoutsBox; late Box _prRecordsBox; late Box _settingsBox; StorageService() { _exercisesBox = Hive.box('exercises'); _workoutsBox = Hive.box('workouts'); _prRecordsBox = Hive.box('pr_records'); _settingsBox = Hive.box('settings'); } static Future initDefaultData() async { final exercisesBox = Hive.box('exercises'); if (exercisesBox.isEmpty) { for (var exercise in ExerciseDatabase.defaultExercises) { await exercisesBox.put(exercise.id, exercise); } } // Inicializuj další data pokud potřeba (např. default settings) } // Exercises methods List getAllExercises() => _exercisesBox.values.toList(); Future addCustomExercise(Exercise exercise) async { exercise.id = const Uuid().v4(); exercise.isCustom = true; await _exercisesBox.put(exercise.id, exercise); notifyListeners(); } Future toggleFavorite(String exerciseId) async { final exercise = _exercisesBox.get(exerciseId); if (exercise != null) { exercise.isFavorite = !exercise.isFavorite; await exercise.save(); notifyListeners(); } } List searchExercises(String query) { final lowerQuery = query.toLowerCase(); return getAllExercises().where((ex) { return ex.name.toLowerCase().contains(lowerQuery) || ex.muscleGroup.toLowerCase().contains(lowerQuery) || ex.category.toLowerCase().contains(lowerQuery); }).toList(); } // Workouts methods Future addWorkout(Workout workout) async { await _workoutsBox.put(workout.id, workout); // Update PRs automatically await _updatePrsFromWorkout(workout); notifyListeners(); } List getWorkoutsByDate(DateTime date) { return _workoutsBox.values.where((w) => w.date.year == date.year && w.date.month == date.month && w.date.day == date.day ).toList(); } // PR methods List getPrsForExercise(String exerciseId) { return _prRecordsBox.values.where((pr) => pr.exerciseId == exerciseId).toList() ..sort((a, b) => b.date.compareTo(a.date)); } Future _updatePrsFromWorkout(Workout workout) async { for (int i = 0; i < workout.exerciseIds.length; i++) { final exerciseId = workout.exerciseIds[i]; final sets = workout.sets[i]; final maxSet = sets.reduce((curr, next) => curr.weight > next.weight ? curr : (curr.weight == next.weight && curr.reps > next.reps ? curr : next)); final existingPrs = getPrsForExercise(exerciseId); final bestPr = existingPrs.isNotEmpty ? existingPrs.first : null; if (bestPr == null || maxSet.weight > bestPr.weight || (maxSet.weight == bestPr.weight && maxSet.reps > bestPr.reps)) { final newPr = PrRecord( id: const Uuid().v4(), exerciseId: exerciseId, date: workout.date, weight: maxSet.weight, reps: maxSet.reps, oneRm: _calculateOneRm(maxSet.weight, maxSet.reps), ); await _prRecordsBox.put(newPr.id, newPr); } } } double _calculateOneRm(double weight, int reps) { // Epley formula return weight * (1 + reps / 30); } // Settings methods (e.g., units, theme) bool get isMetric => _settingsBox.get('units', defaultValue: true) as bool; Future setMetric(bool value) async { await _settingsBox.put('units', value); notifyListeners(); } // Další metody: delete, export, etc. }
